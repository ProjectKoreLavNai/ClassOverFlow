-- ============================================================================
-- 1. LEAST-PRIVILEGE DATABASE ACCOUNTS & PERMISSIONS
-- ============================================================================

DO $$
BEGIN
    -- Administrative / migration role (not used by FastAPI runtime)
    IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = 'classoverflow_admin') THEN
        CREATE ROLE classoverflow_admin NOLOGIN;
    END IF;

    -- FastAPI runtime application role: NO superuser, NO createdb, NO createrole, NO bypassrls
    IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = 'classoverflow_app') THEN
        CREATE ROLE classoverflow_app LOGIN PASSWORD 'REPLACE_WITH_STRONG_ENV_SECRET'
            NOSUPERUSER NOCREATEDB NOCREATEROLE NOINHERIT NOBYPASSRLS;
    END IF;

    -- Safely grant CONNECT using dynamic identifier formatting (prevents syntax error 42601)
    EXECUTE format('GRANT CONNECT ON DATABASE %I TO classoverflow_app;', current_database());
END
$$;

-- Grant least-privilege CRUD access on public schema to FastAPI role
GRANT USAGE ON SCHEMA public TO classoverflow_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO classoverflow_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO classoverflow_app;

-- Ensure future tables/sequences created in public also receive least-privilege CRUD
ALTER DEFAULT PRIVILEGES IN SCHEMA public
    GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO classoverflow_app;
ALTER DEFAULT PRIVILEGES IN SCHEMA public
    GRANT USAGE, SELECT ON SEQUENCES TO classoverflow_app;

-- CRITICAL SECURITY MEASURE: Audit logs are strictly append-only (cannot alter or erase history)
REVOKE UPDATE, DELETE, TRUNCATE ON TABLE audit_logs FROM classoverflow_app;

-- ============================================================================
-- 2. SESSION CONTEXT HELPER FUNCTIONS (PARAMETERIZED FASTAPI CONTEXT)
-- ============================================================================
-- FastAPI runs this parameterized query at the start of every transaction:
-- SELECT set_config('app.current_user_id', :uid, true),
--        set_config('app.current_user_role', :role, true);

CREATE OR REPLACE FUNCTION current_app_user_id() RETURNS INT AS $$
    SELECT NULLIF(current_setting('app.current_user_id', true), '')::INT;
$$ LANGUAGE SQL STABLE;

CREATE OR REPLACE FUNCTION current_app_user_role() RETURNS VARCHAR AS $$
    SELECT NULLIF(current_setting('app.current_user_role', true), '')::VARCHAR;
$$ LANGUAGE SQL STABLE;

-- These SECURITY DEFINER helpers avoid recursive RLS when course and enrollment
-- policies need to check each other's rows. Apply this script as the table owner.
CREATE OR REPLACE FUNCTION public.app_current_user_teaches_course(p_course_id INT)
RETURNS BOOLEAN
LANGUAGE SQL STABLE SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM public.courses c
        WHERE c.course_id = p_course_id
          AND c.teacher_id = public.current_app_user_id()
    );
$$;

CREATE OR REPLACE FUNCTION public.app_current_user_enrolled_in_course(p_course_id INT)
RETURNS BOOLEAN
LANGUAGE SQL STABLE SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM public.enrollments e
        WHERE e.course_id = p_course_id
          AND e.student_id = public.current_app_user_id()
    );
$$;

REVOKE ALL ON FUNCTION public.app_current_user_teaches_course(INT) FROM PUBLIC;
REVOKE ALL ON FUNCTION public.app_current_user_enrolled_in_course(INT) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.app_current_user_teaches_course(INT) TO classoverflow_app;
GRANT EXECUTE ON FUNCTION public.app_current_user_enrolled_in_course(INT) TO classoverflow_app;

-- ----------------------------------------------------------------------------
-- A. USERS, PROFILES & STREAMING PLATFORMS
-- ACM: Users view/manage own profile; Admin views all, verifies accounts, changes roles, deletes accounts
-- ----------------------------------------------------------------------------

DROP POLICY IF EXISTS users_select_policy ON users;
CREATE POLICY users_select_policy ON users FOR SELECT
    USING (current_app_user_role() = 'ADMIN' OR user_id = current_app_user_id());

DROP POLICY IF EXISTS users_admin_manage_policy ON users;
CREATE POLICY users_admin_manage_policy ON users FOR ALL
    USING (current_app_user_role() = 'ADMIN')
    WITH CHECK (current_app_user_role() = 'ADMIN');

DROP POLICY IF EXISTS students_select_policy ON students;
CREATE POLICY students_select_policy ON students FOR SELECT
    USING (
        current_app_user_role() = 'ADMIN'
        OR student_id = current_app_user_id()
        OR current_app_user_role() = 'TEACHER'
    );

DROP POLICY IF EXISTS students_update_own_policy ON students;
CREATE POLICY students_update_own_policy ON students FOR UPDATE
    USING (student_id = current_app_user_id() OR current_app_user_role() = 'ADMIN')
    WITH CHECK (
        current_app_user_role() = 'ADMIN'
        OR (
            student_id = current_app_user_id()
            -- Prevent students from self-elevating cr_status or is_verified!
            AND cr_status = (SELECT s.cr_status FROM students s WHERE s.student_id = current_app_user_id())
            AND is_verified = (SELECT s.is_verified FROM students s WHERE s.student_id = current_app_user_id())
        )
    );

DROP POLICY IF EXISTS students_admin_insert_delete_policy ON students;
CREATE POLICY students_admin_insert_delete_policy ON students FOR ALL
    USING (current_app_user_role() = 'ADMIN')
    WITH CHECK (current_app_user_role() = 'ADMIN');

DROP POLICY IF EXISTS teachers_select_policy ON teachers;
CREATE POLICY teachers_select_policy ON teachers FOR SELECT
    USING (current_app_user_id() IS NOT NULL);

DROP POLICY IF EXISTS teachers_update_own_policy ON teachers;
CREATE POLICY teachers_update_own_policy ON teachers FOR UPDATE
    USING (teacher_id = current_app_user_id() OR current_app_user_role() = 'ADMIN')
    WITH CHECK (
        current_app_user_role() = 'ADMIN'
        OR (
            teacher_id = current_app_user_id()
            -- Prevent teachers from self-verifying their own account
            AND is_verified = (SELECT t.is_verified FROM teachers t WHERE t.teacher_id = current_app_user_id())
        )
    );

DROP POLICY IF EXISTS teachers_admin_manage_policy ON teachers;
CREATE POLICY teachers_admin_manage_policy ON teachers FOR ALL
    USING (current_app_user_role() = 'ADMIN')
    WITH CHECK (current_app_user_role() = 'ADMIN');

DROP POLICY IF EXISTS admins_rls_policy ON admins;
CREATE POLICY admins_rls_policy ON admins FOR ALL
    USING (current_app_user_role() = 'ADMIN')
    WITH CHECK (current_app_user_role() = 'ADMIN');

-- Streaming platforms: Students/CRs manage own settings; Teachers/Admins can view status
DROP POLICY IF EXISTS user_platforms_select_policy ON user_platforms;
CREATE POLICY user_platforms_select_policy ON user_platforms FOR SELECT
    USING (
        student_id = current_app_user_id()
        OR current_app_user_role() IN ('TEACHER', 'ADMIN')
    );

DROP POLICY IF EXISTS user_platforms_write_policy ON user_platforms;

    WITH CHECK (current_app_user_role() = 'STUDENT' AND student_id = current_app_user_id());

CREATE POLICY announcements_select_policy ON announcements FOR SELECT

    USING (
        public.current_app_user_role() = 'TEACHER'
        AND posted_by = public.current_app_user_id()

    WITH CHECK (

        AND public.app_current_user_teaches_course(course_id)
    );

DROP POLICY IF EXISTS comments_select_policy ON announcement_comments;
CREATE POLICY comments_select_policy ON announcement_comments FOR SELECT
    USING (
        current_app_user_role() <> 'ADMIN'
        AND EXISTS (
            SELECT 1 FROM public.announcements a
            WHERE a.announcement_id = announcement_comments.announcement_id
              AND (
                  public.app_current_user_teaches_course(a.course_id)
                  OR public.app_current_user_enrolled_in_course(a.course_id)
              )
        )
    );

DROP POLICY IF EXISTS comments_insert_policy ON announcement_comments;
CREATE POLICY comments_insert_policy ON announcement_comments FOR INSERT
    WITH CHECK (
        user_id = current_app_user_id()
        AND (
            EXISTS (
                SELECT 1 FROM announcements a
                JOIN courses c ON c.course_id = a.course_id
                WHERE a.announcement_id = announcement_comments.announcement_id
                  AND c.teacher_id = current_app_user_id()
            )
            OR EXISTS (
                SELECT 1 FROM announcements a
                JOIN enrollments e ON e.course_id = a.course_id
                WHERE a.announcement_id = announcement_comments.announcement_id
                  AND e.student_id = current_app_user_id()
            )
        )
    );

DROP POLICY IF EXISTS comments_delete_policy ON announcement_comments;
CREATE POLICY comments_delete_policy ON announcement_comments FOR DELETE
    USING (
        user_id = current_app_user_id()
        OR EXISTS (
            SELECT 1 FROM announcements a
            JOIN courses c ON c.course_id = a.course_id
            WHERE a.announcement_id = announcement_comments.announcement_id
              AND c.teacher_id = current_app_user_id()
        )
    );

-- Notification Queue: Accessible by Admin, Course Teacher, or Enrolled CR triggering schedule/deadline alerts
DROP POLICY IF EXISTS notification_queue_rls_policy ON notification_queue;
CREATE POLICY notification_queue_rls_policy ON notification_queue FOR ALL
    USING (
        current_app_user_role() = 'ADMIN'
        OR EXISTS (
            SELECT 1 FROM courses c
            WHERE c.course_id = notification_queue.course_id AND c.teacher_id = current_app_user_id()
        )
        OR (
            current_app_user_role() = 'STUDENT'
            AND triggered_by_user = current_app_user_id()
            AND EXISTS (
                SELECT 1 FROM enrollments e
                JOIN students s ON s.student_id = e.student_id
                WHERE e.course_id = notification_queue.course_id
                  AND s.student_id = current_app_user_id()
                  AND s.cr_status = TRUE
            )
        )
    )
    WITH CHECK (
        current_app_user_role() = 'ADMIN'
        OR EXISTS (
            SELECT 1 FROM courses c
            WHERE c.course_id = notification_queue.course_id AND c.teacher_id = current_app_user_id()
        )
        OR (
            current_app_user_role() = 'STUDENT'
            AND triggered_by_user = current_app_user_id()
            AND event_type IN ('SCHEDULE_UPDATE', 'DEADLINE_ALERT')
            AND EXISTS (
                SELECT 1 FROM enrollments e
                JOIN students s ON s.student_id = e.student_id
                WHERE e.course_id = notification_queue.course_id
                  AND s.student_id = current_app_user_id()
                  AND s.cr_status = TRUE
            )
        )
    );

-- Submissions: Strict isolation per student (Admin has NO access per ACM)
DROP POLICY IF EXISTS submissions_select_policy ON submissions;
CREATE POLICY submissions_select_policy ON submissions FOR SELECT
    USING (
        (current_app_user_role() = 'STUDENT' AND student_id = current_app_user_id())
        OR (current_app_user_role() = 'TEACHER' AND EXISTS (
            SELECT 1 FROM assignments a
            JOIN courses c ON c.course_id = a.course_id
            WHERE a.assignment_id = submissions.assignment_id AND c.teacher_id = current_app_user_id()
        ))
    );

-- Student can insert a new submission attempt ONLY before the deadline for an enrolled course
DROP POLICY IF EXISTS submissions_student_insert_policy ON submissions;
CREATE POLICY submissions_student_insert_policy ON submissions FOR INSERT
    WITH CHECK (
        current_app_user_role() = 'STUDENT'
        AND student_id = current_app_user_id()
        AND status = 'SUBMITTED'
        AND marks_obtained IS NULL
        AND EXISTS (
            SELECT 1 FROM assignments a
            JOIN enrollments e ON e.course_id = a.course_id
            WHERE a.assignment_id = submissions.assignment_id
              AND e.student_id = current_app_user_id()
              AND (a.deadline IS NULL OR NOW() <= a.deadline)
        )
    );

-- Student can unsubmit their own ungraded submission ONLY BEFORE DEADLINE; Teacher can grade at any time
DROP POLICY IF EXISTS submissions_update_policy ON submissions;
CREATE POLICY submissions_update_policy ON submissions FOR UPDATE
    USING (
        (current_app_user_role() = 'STUDENT'
            AND student_id = current_app_user_id()
            AND status = 'SUBMITTED'
            AND marks_obtained IS NULL
        )
        OR (current_app_user_role() = 'TEACHER' AND EXISTS (
            SELECT 1 FROM assignments a
            JOIN courses c ON c.course_id = a.course_id
            WHERE a.assignment_id = submissions.assignment_id AND c.teacher_id = current_app_user_id()
        ))
    )
    WITH CHECK (
        (current_app_user_role() = 'STUDENT'
            AND student_id = current_app_user_id()
            AND status = 'UNSUBMITTED'
            AND marks_obtained IS NULL
            AND EXISTS (
                SELECT 1 FROM assignments a
                WHERE a.assignment_id = submissions.assignment_id
                  AND (a.deadline IS NULL OR NOW() <= a.deadline)
            )
        )
        OR (current_app_user_role() = 'TEACHER' AND EXISTS (
            SELECT 1 FROM assignments a
            JOIN courses c ON c.course_id = a.course_id
            WHERE a.assignment_id = submissions.assignment_id AND c.teacher_id = current_app_user_id()
        ))
    );

-- E. PRIVATE FILES, CLASS MATERIALS, TOPICS, LECTURE PLANS & LECTURE NOTES
-- ----------------------------------------------------------------------------

DROP POLICY IF EXISTS files_select_policy ON files;
CREATE POLICY files_select_policy ON files FOR SELECT
    USING (
        -- Uploaders can always view their own files
        uploaded_by_student = current_app_user_id()
        OR uploaded_by_teacher = current_app_user_id()
        -- Admin can view course materials/attachments (not private student submissions per ACM)
        OR (current_app_user_role() = 'ADMIN' AND category <> 'Assignment_submission')
        -- Assigned Teacher can view all files in their courses (including student submissions)
        OR (current_app_user_role() = 'TEACHER' AND EXISTS (
            SELECT 1 FROM courses c WHERE c.course_id = files.course_id AND c.teacher_id = current_app_user_id()
        ))
        -- Enrolled Students can view course materials, announcement attachments, and assignment attachments
        OR (current_app_user_role() = 'STUDENT'
            AND category IN ('Assignment_attachment', 'Announcement_attachment', 'Class_material')
            AND EXISTS (
                SELECT 1 FROM enrollments e WHERE e.course_id = files.course_id AND e.student_id = current_app_user_id()
            )
        )
    );

DROP POLICY IF EXISTS files_insert_policy ON files;
CREATE POLICY files_insert_policy ON files FOR INSERT
    WITH CHECK (
        -- Student uploading a file for their own active submission before deadline
        (current_app_user_role() = 'STUDENT'
            AND uploaded_by_student = current_app_user_id()
            AND category = 'Assignment_submission'
            AND EXISTS (
                SELECT 1 FROM submissions s
                JOIN assignments a ON a.assignment_id = s.assignment_id
                WHERE s.submission_id = files.submission_id
                  AND s.student_id = current_app_user_id()
                  AND s.status = 'SUBMITTED'
                  AND (a.deadline IS NULL OR NOW() <= a.deadline)
            )
        )
        -- Teacher uploading course materials or attachments for their own course
        OR (current_app_user_role() = 'TEACHER'
            AND uploaded_by_teacher = current_app_user_id()
            AND category IN ('Assignment_attachment', 'Announcement_attachment', 'Class_material')
            AND EXISTS (
                SELECT 1 FROM courses c
                WHERE c.course_id = files.course_id AND c.teacher_id = current_app_user_id()
            )
        )
    );

DROP POLICY IF EXISTS files_delete_policy ON files;
CREATE POLICY files_delete_policy ON files FOR DELETE
    USING (
        current_app_user_role() = 'ADMIN'
        OR (current_app_user_role() = 'TEACHER' AND uploaded_by_teacher = current_app_user_id())
        OR (current_app_user_role() = 'STUDENT'
            AND uploaded_by_student = current_app_user_id()
            AND EXISTS (
                SELECT 1 FROM submissions s
                WHERE s.submission_id = files.submission_id
                  AND s.student_id = current_app_user_id()
                  AND s.status <> 'GRADED'
            )
        )
    );

-- Trigger 1: Audit Marks Insert / Update / Delete
CREATE OR REPLACE FUNCTION fn_audit_marks_changes() RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit_logs (actor_user_id, action_type, target_table, target_record_id, new_values)
        VALUES (
            current_app_user_id(), 'MARKS_INSERT', 'marks', NEW.mark_id,
            jsonb_build_object(
                'student_id', NEW.student_id,
                'course_id', NEW.course_id,
                'assessment_name', NEW.assessment_name,
                'marks_obtained', NEW.marks_obtained,
                'total_marks', NEW.total_marks
            )
        );
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' AND (OLD.marks_obtained IS DISTINCT FROM NEW.marks_obtained) THEN
        INSERT INTO audit_logs (actor_user_id, action_type, target_table, target_record_id, old_values, new_values)
        VALUES (
            current_app_user_id(), 'MARKS_UPDATE', 'marks', NEW.mark_id,
            jsonb_build_object('marks_obtained', OLD.marks_obtained),
            jsonb_build_object('marks_obtained', NEW.marks_obtained)
        );
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO audit_logs (actor_user_id, action_type, target_table, target_record_id, old_values)
        VALUES (
            current_app_user_id(), 'MARKS_DELETE', 'marks', OLD.mark_id,
            jsonb_build_object(
                'student_id', OLD.student_id,
                'assessment_name', OLD.assessment_name,
                'marks_obtained', OLD.marks_obtained
            )
        );
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS trg_audit_marks ON marks;
CREATE TRIGGER trg_audit_marks
AFTER INSERT OR UPDATE OR DELETE ON marks
FOR EACH ROW EXECUTE FUNCTION fn_audit_marks_changes();

-- 6. SUPABASE STORAGE BUCKETS & STORAGE RLS POLICIES (MALICIOUS UPLOAD PROTECTION)
-- ============================================================================
-- Automatically creates/updates the two private buckets on Supabase with a 25MB limit
-- and strict MIME-type whitelist, and blocks direct anonymous public access.

DO $$
BEGIN
    IF EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema = 'storage' AND table_name = 'buckets') THEN
        INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
        VALUES
            (
                'private-course-files',
                'private-course-files',
                FALSE,
                26214400, -- 25MB max
                ARRAY[
                    'application/pdf',
                    'image/jpeg',
                    'image/png',
                    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
                    'application/vnd.openxmlformats-officedocument.presentationml.presentation',
                    'application/zip'
                ]
            ),
            (
                'private-submissions',
                'private-submissions',
                FALSE,
                26214400, -- 25MB max
                ARRAY[
                    'application/pdf',
                    'image/jpeg',
                    'image/png',
                    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
                    'application/vnd.openxmlformats-officedocument.presentationml.presentation',
                    'application/zip'
                ]
            )
        ON CONFLICT (id) DO UPDATE
        SET public = FALSE,
            file_size_limit = EXCLUDED.file_size_limit,
            allowed_mime_types = EXCLUDED.allowed_mime_types;
    END IF;
END
$$;

-- Storage Object Policies (Blocks anonymous access; files are served via FastAPI short-lived signed URLs)
DROP POLICY IF EXISTS "Deny Public Anon Access" ON storage.objects;
CREATE POLICY "Deny Public Anon Access" ON storage.objects
    FOR ALL
    TO anon
    USING (FALSE)
    WITH CHECK (FALSE);

DROP POLICY IF EXISTS "Allow Authenticated Backend Storage Operations" ON storage.objects;
CREATE POLICY "Allow Authenticated Backend Storage Operations" ON storage.objects
    FOR ALL
    TO authenticated
    USING (bucket_id IN ('private-course-files', 'private-submissions'))
    WITH CHECK (bucket_id IN ('private-course-files', 'private-submissions'));
